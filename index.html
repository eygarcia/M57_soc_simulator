    Live Bash Automation Demo
    This example turns raw web-log records into a repeatable top-host report. The same script can later be scheduled with cron.
    #!/usr/bin/env bash
LOG="access.log"
REPORT="top_hosts_$(date +%F).txt"

cut -d' ' -f1 "$LOG" |
  sort |
  uniq -c |
  sort -rn > "$REPORT"

echo "Report saved: $REPORT"
